import channels from '../../data/channels.json';
import layers from '../../data/layers.json';

export const revenueLayer = layers[0];
export const netLayer = layers[4];

export const margins = revenueLayer.channels.map(ch => {
  const net = netLayer.channels.find(n => n.channel_name === ch.channel_name);
  return {
    channel_name: ch.channel_name,
    revenue: ch.value,
    contribution: net?.value || 0,
    margin_pct: net ? ((net.value / ch.value) * 100).toFixed(1) : '0.0',
    erosion: ch.value - (net?.value || 0),
  };
}).sort((a, b) => b.revenue - a.revenue);

export const totalRevenue = margins.reduce((s, m) => s + m.revenue, 0);
export const totalContribution = margins.reduce((s, m) => s + m.contribution, 0);
export const totalErosion = totalRevenue - totalContribution;

## The full picture

Four layers of cost — COGS, trade deductions, compliance fines, and operational overhead — separate gross revenue from actual contribution. Across all channels, {new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 1, notation: 'compact' }).format(totalErosion)} disappears between the top line and the bottom line. That is {((totalErosion / totalRevenue) * 100).toFixed(1)}% of revenue.

But the erosion is not uniform. The table below reveals which channels retain the most — and least — of every dollar earned.

The standout: DTC retains just 71.6% of revenue — the worst margin in the portfolio. Its zero deductions and zero fines cannot compensate for a COGS ratio (28.4%) that is triple the retail average (3.6%). Every dollar shifted from retail to DTC expecting better margins would, on these numbers, destroy value.
