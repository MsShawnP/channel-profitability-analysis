import channels from '../../data/channels.json';
import layers from '../../data/layers.json';
import MarginTable from '../../components/charts/MarginTable';

export const revenueLayer = layers[0];
export const netLayer = layers[4];

export const buildMargins = (type) => revenueLayer.channels
  .filter(ch => ch.channel_type === type)
  .map(ch => {
    const net = netLayer.channels.find(n => n.channel_name === ch.channel_name);
    return {
      channel_name: ch.channel_name,
      revenue: ch.value,
      contribution: net?.value || 0,
      margin_pct: net ? ((net.value / ch.value) * 100).toFixed(1) : '0.0',
      erosion: ch.value - (net?.value || 0),
    };
  }).sort((a, b) => b.revenue - a.revenue);

export const retailMargins = buildMargins('retailer');
export const distMargins = buildMargins('distributor');
export const dtcMargins = buildMargins('DTC');
export const allMargins = [...retailMargins, ...distMargins, ...dtcMargins];

export const totalRevenue = allMargins.reduce((s, m) => s + m.revenue, 0);
export const totalContribution = allMargins.reduce((s, m) => s + m.contribution, 0);
export const totalErosion = totalRevenue - totalContribution;

export const fmt = (v) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0, notation: 'compact' }).format(v);

## The full picture

Four layers of cost — COGS, trade deductions, compliance fines, and operational overhead — separate gross revenue from actual contribution. Across all channels, {fmt(totalErosion)} disappears between the top line and the bottom line. That is {((totalErosion / totalRevenue) * 100).toFixed(1)}% of revenue.

But the erosion is not uniform, and it differs radically by segment. The cost stack that hits wholesale channels barely touches DTC.

<p class="segment-label">Retailers</p>

<MarginTable rows={retailMargins} />

Retail contribution margins range from the low twenties (Walmart, Costco) to the mid-thirties (Kroger, Sprouts). COGS is the primary driver — consuming a significant share of revenue before any deductions begin. After trade deductions and compliance fines strip additional margin, the large nationals retain barely a quarter of each revenue dollar. The mid-tier retailers (Kroger, Sprouts, Regional Group) outperform on margin — lower deduction rates and minimal fines compensate for their smaller scale.

<p class="segment-label">Distributors</p>

<MarginTable rows={distMargins} />

UNFI and KeHE retain 22.3% and 23.1% respectively — the lowest margins in the wholesale portfolio. Distributor COGS runs 60% of revenue (higher than retail), and spoilage fines add a cost layer that retailers largely avoid. Together the distributors represent $6.4M in revenue but contribute only $1.5M after costs.

<p class="segment-label">Direct-to-Consumer</p>

<MarginTable rows={dtcMargins} highlight="DTC" />

DTC retains 73.2% of revenue — the highest margin in the portfolio by a wide margin. Its COGS ratio (26.8%) is less than half the wholesale average because it sells at full retail price. Zero deductions and zero fines leave the gross margin intact. Every dollar of DTC revenue contributes nearly three times what a wholesale dollar does.

<p class="chart-footnote">Margin = net contribution / gross revenue. Erosion = gross revenue minus net contribution. DTC's lower COGS ratio reflects full retail pricing; wholesale channels sell at wholesale prices, compressing margins before deductions begin.</p>
