import channels from '../../data/channels.json';
import layers from '../../data/layers.json';
import MarginTable from '../../components/charts/MarginTable';

export const revenueLayer = layers[0];
export const netLayer = layers[4];

export const buildMargins = (type) => revenueLayer.channels
  .filter(ch => ch.channel_type === type)
  .map(ch => {
    const net = netLayer.channels.find(n => n.channel_name === ch.channel_name);
    return {
      channel_name: ch.channel_name,
      revenue: ch.value,
      contribution: net?.value || 0,
      margin_pct: net ? ((net.value / ch.value) * 100).toFixed(1) : '0.0',
      erosion: ch.value - (net?.value || 0),
    };
  }).sort((a, b) => b.revenue - a.revenue);

export const retailMargins = buildMargins('retailer');
export const distMargins = buildMargins('distributor');
export const dtcMargins = buildMargins('DTC');
export const allMargins = [...retailMargins, ...distMargins, ...dtcMargins];

export const totalRevenue = allMargins.reduce((s, m) => s + m.revenue, 0);
export const totalContribution = allMargins.reduce((s, m) => s + m.contribution, 0);
export const totalErosion = totalRevenue - totalContribution;

export const fmt = (v) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0, notation: 'compact' }).format(v);

## The full picture

Four layers of cost — COGS, trade deductions, compliance fines, and operational overhead — separate gross revenue from actual contribution. Across all channels, {fmt(totalErosion)} disappears between the top line and the bottom line. That is {((totalErosion / totalRevenue) * 100).toFixed(1)}% of revenue.

But the erosion is not uniform, and it differs by segment. Retail channels returned $53,000 more per million deployed than distribution.

<p class="segment-label">Retailers</p>

<MarginTable rows={retailMargins} />

Retail channels produced a 50.6% contribution margin after COGS and trade deductions. Distribution came in at 45.3% — a 5.3-point gap. The conventional assumption that distribution is the cheaper route did not survive the cost accounting.

<p class="segment-label">Distributors</p>

<MarginTable rows={distMargins} />

UNFI, KeHE, and DPI Northwest produced a 45.3% contribution margin after COGS and trade deductions — below the retail average of 50.6%. Distributor COGS runs lower as a percentage of revenue, but the margin advantage does not survive the full cost stack. Together the three distributors represent $8.2M/yr in revenue.

<p class="segment-label">Direct-to-Consumer</p>

<MarginTable rows={dtcMargins} highlight="DTC" />

DTC retains 82.6% of revenue — better than both retailers and distributors on a per-dollar basis. Its COGS ratio (17.4%) is comparable to retailers, not distributors, because DTC sells at full retail pricing on the same product cost base. With zero post-invoice deductions, zero fines, and zero dispute overhead, the margin advantage comes entirely from avoiding the deduction stack.

<p class="chart-footnote">Margin = net contribution / gross revenue. Erosion = gross revenue minus net contribution. DTC's higher COGS ratio reflects full retail pricing on the same product cost; wholesale channels sell at wholesale prices, spreading the fixed product cost over higher volumes.</p>
